114. 绝对寻址 ROR

添加 ROR 绝对寻址操作码 ``0x6E``。

第 114 / 356 · tests/chapter_01_cpu/test_114_ROR_absolute.py

上一课: 113 下一课: 115

在这一步中，使用之前课程中的指令与寻址导入，只在 `emulator/cpu/opcodes.py` 中添加绝对寻址的处理函数和表条目。

为什么需要这一步

ROR 需要一种完整地址的内存形式，使程序能够旋转零页之外的字节，同时复用同一个读-改-写指令原语。

建议实现

def ror_absolute(cpu: CPU):
    addr = absolute(cpu)
    ror(cpu, addr)

OPCODE_TABLE = {
    ...
    0x6E: ror_absolute,
}

`emulator/cpu/addressing_modes.py::absolute 调用 CPU.fetch_word，因此先解码低字节再解码高字节。6E 00 02 通过 instructions.ror 把 $0200` 处的字节连同 Carry 一起旋转。从概念上看内存被读/改/写了一次，A 不变，C/Z/N 遵循旋转结果，操作码加上两字节操作数使 PC 前进三个字节。

误解澄清：`00 02 不是地址 $0002，其本身也不是要旋转的数据。本课不涉及的内容：加入 X（第 115 课）、修改 absolute，或者在处理函数里重新实现 ror`。

import inspect

from emulator.bus.cpu_bus import CpuBus
from emulator.cpu import opcodes
from emulator.cpu.cpu import CPU
from emulator.memory.fake_rom import FakeROM


def make_cpu_with_rom():
    rom = FakeROM()
    rom.write(0x7FFC, 0x00)
    rom.write(0x7FFD, 0x80)
    bus = CpuBus(program_rom=rom)
    return CPU(bus), bus, rom


def test_ror_absolute_handler_exists_and_is_in_opcode_table():
    """Objective: create ror_absolute(cpu) and add 0x6E to OPCODE_TABLE."""
    assert hasattr(opcodes, "ror_absolute")
    assert callable(opcodes.ror_absolute)
    assert list(inspect.signature(opcodes.ror_absolute).parameters) == ["cpu"]
    assert opcodes.OPCODE_TABLE[0x6E] is opcodes.ror_absolute


def test_opcode_6E_ror_absolute_rotates_memory_value():
    """Objective: 6E 00 02 means ROR $0200, so RAM[$0200] is rotated."""
    cpu, bus, rom = make_cpu_with_rom()
    rom.write(0x0000, 0x6E)
    rom.write(0x0001, 0x00)
    rom.write(0x0002, 0x02)
    bus.write(0x0200, 0b0000_0110)

    cpu.reset()
    cpu.step()

    assert bus.read(0x0200) == 0b0000_0011
    assert cpu.pc == 0x8003
