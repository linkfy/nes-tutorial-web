182. Jmp absolute

подключить опкод JMP absolute $4C.

Предусловие: на шаге 181 была добавлена `jmp. На этом шаге внесите следующие изменения в emulator/cpu/opcodes.py` (включая импорт и запись в таблице):

from emulator.cpu.instructions import jmp
from emulator.cpu.addressing_modes import absolute

def jmp_absolute(cpu: CPU):
    addr = absolute(cpu)
    jmp(cpu, addr)

OPCODE_TABLE = {
    # existing entries...
    0x4C: jmp_absolute,
}

Опкод

0x4C -> JMP $hhhh

Цель

создать jmp_absolute(cpu), использовать absolute(cpu), затем jmp(cpu, addr).

Рекомендации

JMP absolute отличается от LDA absolute.

Для LDA $1234:

A = memory[$1234]

Для JMP $1234:

PC = $1234

Поэтому обработчик опкода не должен читать память по целевому адресу. Операнд absolute уже является новым значением PC.

Зачем нужен этот шаг

`absolute считывает два операндных байта в порядке little-endian и возвращает их 16-битное значение; jmp присваивает его. Инварианты: один шаг потребляет три байта перед заменой PC, сохраняет флаги/регистры/стек/память и отображает только $4C. Заблуждение: в отличие от обработчиков загрузки, этот обработчик не должен вызывать cpu.bus.read(addr)`.

За рамками шага: JMP indirect и опкод $6C — шаг 183; JSR и его влияние на стек — шаги 184–185.

import inspect

from emulator.bus.cpu_bus import CpuBus
from emulator.cpu import opcodes
from emulator.cpu.cpu import CPU
from emulator.memory.fake_rom import FakeROM


def make_cpu_with_rom():
    rom = FakeROM()
    rom.write(0x7FFC, 0x00)
    rom.write(0x7FFD, 0x80)
    bus = CpuBus(program_rom=rom)
    return CPU(bus), bus, rom


def test_jmp_absolute_handler_exists_and_is_in_opcode_table():
    """Objective: create jmp_absolute(cpu) and add 0x4C to OPCODE_TABLE."""
    assert hasattr(opcodes, "jmp_absolute")
    assert callable(opcodes.jmp_absolute)
    assert list(inspect.signature(opcodes.jmp_absolute).parameters) == ["cpu"]
    assert opcodes.OPCODE_TABLE[0x4C] is opcodes.jmp_absolute


def test_opcode_4C_jmp_absolute_sets_pc_to_operand_address():
    """Objective: 4C 34 12 means JMP $1234, so PC becomes $1234."""
    cpu, bus, rom = make_cpu_with_rom()
    rom.write(0x0000, 0x4C)
    rom.write(0x0001, 0x34)
    rom.write(0x0002, 0x12)

    cpu.reset()
    cpu.step()

    assert cpu.pc == 0x1234


def test_opcode_4C_jmp_absolute_does_not_read_target_memory_as_new_pc():
    """
    Objective:
    JMP $1234 jumps to $1234 directly.

    It must not do:
        PC = memory[$1234]
    """
    cpu, bus, rom = make_cpu_with_rom()
    rom.write(0x0000, 0x4C)
    rom.write(0x0001, 0x34)
    rom.write(0x0002, 0x12)
    bus.write(0x1234, 0x99)

    cpu.reset()
    cpu.step()

    assert cpu.pc == 0x1234
